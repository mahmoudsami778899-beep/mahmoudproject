# ممكن احط اف وايلس بردو بس حبيت اصغر الكود 







students = [
    {"name": "ali", "grades": [85, 90, 80]},
    
    {"name": "sara", "grades": [90,100,95]},
    
    {"name": "aya", "grades": [70, 75, 80]}
]


for student in students:
    
    average = sum(student["grades"]) / len(student["grades"])
    
    print(f"{student['name']} - Average Grade: {average}")
